### B. React / Next.js Modern Component
If your website uses React or Next.js:

```jsx
import React, { useState } from 'react';
import workflowData from './portfolio_workflows_data.json';

export default function N8nPortfolioSection() {
  const [filter, setFilter] = useState('All');
  const categories = ['All', 'AI Chatbots & Conversational Commerce', 'Lead Generation & CRM Automation', 'Productivity & Content ETL Pipelines'];

  const filteredProjects = filter === 'All' 
    ? workflowData.featured_projects 
    : workflowData.featured_projects.filter(p => p.category.includes(filter) || filter.includes(p.category));

  return (
    <section className="py-20 bg-slate-900 text-white" id="workflows">
      <div className="max-w-7xl mx-auto px-6">
        <div className="flex flex-col md:flex-row md:items-end justify-between mb-12">
          <div>
            <span className="text-red-400 font-mono text-sm tracking-wider uppercase">Enterprise Automation</span>
            <h2 className="text-4xl font-bold mt-2">74 Production n8n Workflows</h2>
            <p className="text-slate-400 mt-2 max-w-xl">
              Scalable event-driven pipelines, LangChain Multi-Agents, Qdrant Vector Stores, and relational persistence.
            </p>
          </div>
          <div className="flex flex-wrap gap-2 mt-6 md:mt-0">
            {categories.map(cat => (
              <button
                key={cat}
                onClick={() => setFilter(cat)}
                className={`px-3 py-1.5 rounded-lg text-xs font-medium transition ${
                  filter === cat ? 'bg-red-500 text-white' : 'bg-slate-800 text-slate-400 hover:text-white'
                }`}
              >
                {cat.replace(' & Conversational Commerce', '').replace(' & CRM Automation', '').replace(' & Content ETL Pipelines', '')}
              </button>
            ))}
          </div>
        </div>

        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6">
          {filteredProjects.map(item => (
            <div key={item.id} className="bg-slate-800/80 border border-slate-700/80 rounded-xl p-6 flex flex-col justify-between hover:border-red-500/50 transition duration-300">
              <div>
                <span className="text-xs font-medium text-red-400 block mb-2">{item.category}</span>
                <h3 className="text-lg font-bold text-slate-100 mb-2">{item.title}</h3>
                <p className="text-sm text-slate-300 mb-4 leading-relaxed">{item.github_description}</p>
                <div className="flex flex-wrap gap-1.5 mb-6">
                  {item.tech_stack.map(tech => (
                    <span key={tech} className="text-xs bg-slate-900/60 text-slate-300 px-2 py-0.5 rounded border border-slate-700">
                      {tech}
                    </span>
                  ))}
                </div>
              </div>
              <a 
                href={item.github_url} 
                target="_blank" 
                rel="noreferrer"
                className="w-full py-2.5 text-center text-sm font-semibold rounded-lg bg-slate-700 hover:bg-red-500 text-white transition duration-200"
              >
                View Repository & Architecture &rarr;
              </a>
            </div>
          ))}
        </div>
      </div>
    </section>
  );
}
```
